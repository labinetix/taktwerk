raw-pi

A PI controller from a plain C library, multiple instances by handle.

pi.h:

/* A PI controller over n channels, one state per instance behind a handle. */
#ifndef PI_H
#define PI_H

typedef struct {
    double kp;
    double ki;
} pi_gains;

int pi_init(void **handle, int n, double dt);
int pi_step(void *handle, const pi_gains *gains, const double *sp, const double *pv, double *u);
void pi_terminate(void *handle);

#endif

pi.c:

#include <stdlib.h>
#include "pi.h"

typedef struct {
    int n;
    double dt;
    double integral[];
} pi_state;

int pi_init(void **handle, int n, double dt) {
    pi_state *s;
    int i;
    if (n < 1 || dt <= 0.0) return -1;
    s = malloc(sizeof *s + sizeof(double) * (size_t)n);
    if (!s) return -2;
    s->n = n;
    s->dt = dt;
    for (i = 0; i < n; i++) s->integral[i] = 0.0;
    *handle = s;
    return 0;
}

int pi_step(void *handle, const pi_gains *gains, const double *sp, const double *pv, double *u) {
    pi_state *s = handle;
    int i;
    if (!s) return -1;
    for (i = 0; i < s->n; i++) {
        double e = sp[i] - pv[i];
        s->integral[i] += gains->ki * e * s->dt;
        u[i] = gains->kp * e + s->integral[i];
    }
    return 0;
}

void pi_terminate(void *handle) {
    free(handle);
}

lowpass.h:

/* A bank of first-order low-pass filters over n channels, in the recommended shape. */
#ifndef LOWPASS_H
#define LOWPASS_H

typedef struct {
    int n;              /* 1..64 = 3 */
} lowpass_dims;

typedef struct {
    const double *tau;  /* [n] s: time constant */
    const double *y0;   /* [n]: initial state */
} lowpass_params;

typedef struct {
    const double *u;    /* [n]: filter input */
} lowpass_inputs;

typedef struct {
    double gain;        /* output gain */
} lowpass_tunables;

typedef struct {
    double *y;          /* [n]: gain times the filter state */
} lowpass_outputs;

int lowpass_init(void **h, const lowpass_dims *dims, const lowpass_params *params,
                 double step_size);
int lowpass_step(void *h, double time, const lowpass_inputs *in,
                 const lowpass_tunables *tun, lowpass_outputs *out);
int lowpass_terminate(void *h);

#endif

lowpass.c:

#include <stdlib.h>
#include "lowpass.h"

typedef struct {
    int n;
    double *alpha; /* [n], dt / (tau + dt) */
    double *x;     /* [n], filter state */
} lowpass_state;

int lowpass_init(void **h, const lowpass_dims *dims, const lowpass_params *params,
                 double step_size) {
    lowpass_state *s;
    int i;
    if (dims->n < 1 || step_size <= 0.0) return 1;
    s = calloc(1, sizeof *s);
    if (!s) return 2;
    s->n = dims->n;
    s->alpha = calloc((size_t)s->n, sizeof *s->alpha);
    s->x = calloc((size_t)s->n, sizeof *s->x);
    if (!s->alpha || !s->x) {
        lowpass_terminate(s);
        return 2;
    }
    for (i = 0; i < s->n; i++) {
        if (params->tau[i] < 0.0) {
            lowpass_terminate(s);
            return 3;
        }
        s->alpha[i] = step_size / (params->tau[i] + step_size);
        s->x[i] = params->y0[i];
    }
    *h = s;
    return 0;
}

int lowpass_step(void *h, double time, const lowpass_inputs *in,
                 const lowpass_tunables *tun, lowpass_outputs *out) {
    lowpass_state *s = h;
    int i;
    (void)time;
    for (i = 0; i < s->n; i++) {
        s->x[i] += s->alpha[i] * (in->u[i] - s->x[i]);
        out->y[i] = tun->gain * s->x[i];
    }
    return 0;
}

int lowpass_terminate(void *h) {
    lowpass_state *s = h;
    if (s) {
        free(s->alpha);
        free(s->x);
        free(s);
    }
    return 0;
}

raw-single-entry

io_model.h:

/* A model behind a single entry point: one function for init and step, called with an id and
 * two opaque byte pointers that the library casts to its input and output structs. The library
 * keeps no state; every call re-reads the sizes it reported from the caller's output struct. */
#ifndef IO_MODEL_H
#define IO_MODEL_H

#include <stdint.h>

#define IO_ID 7

#define IO_OK 0
#define IO_NULL_ARGUMENT (-1)
#define IO_UNKNOWN_VARIANT (-2)
#define IO_UNKNOWN_ID (-3)
#define IO_NOT_INITIALISED (-4)
#define IO_BAD_FLAG (-5)

/* What a successful call writes into `ready`: a true byte that is not 1. */
#define IO_TRUE_BYTE 2

typedef struct {
    int32_t *init_flag; /* 1 on the first call (init), 0 on every step */
    int32_t variant;    /* by value between two pointers: which size set to report */
    double *k;          /* scalar gain */
    double *u;          /* nu inputs */
    int32_t *counter;   /* step counter, written back by the library */
    char *title;        /* text the caller supplies, NUL-terminated */
} io_input;

typedef struct {
    char *label;      /* the title echoed back, or the library's own name if empty */
    double *y;        /* ny outputs: y = k * gain * u */
    double *gain;     /* ny x nu, row-major, filled at init */
    double *p;        /* np parameters, filled at init */
    char *names;      /* names written at init */
    int32_t *dim_nu;  /* reported at init */
    int32_t *dim_ny;  /* reported at init */
    int32_t *dim_np;  /* reported at init */
    uint8_t *ready;   /* IO_TRUE_BYTE after every successful call */
} io_output;

int io_call(int id, char *in, char *out);

#endif

io_model.c:

#include "io_model.h"

#include <stddef.h>
#include <string.h>

static int sizes(int32_t variant, int32_t *nu, int32_t *ny, int32_t *np) {
    switch (variant) {
    case 1: *nu = 2; *ny = 3; *np = 4; return 0;
    case 2: *nu = 1; *ny = 1; *np = 2; return 0;
    default: return -1;
    }
}

static void copy_text(char *dst, size_t cap, const char *src) {
    size_t n = strlen(src);
    if (n >= cap) n = cap - 1;
    memcpy(dst, src, n);
    dst[n] = '\0';
}

int io_call(int id, char *in_bytes, char *out_bytes) {
    if (id != IO_ID) return IO_UNKNOWN_ID;
    if (in_bytes == NULL || out_bytes == NULL) return IO_NULL_ARGUMENT;
    io_input *in = (io_input *)in_bytes;
    io_output *out = (io_output *)out_bytes;
    if (in->init_flag == NULL || in->k == NULL || in->u == NULL || in->counter == NULL ||
        out->y == NULL || out->gain == NULL || out->p == NULL || out->dim_nu == NULL ||
        out->dim_ny == NULL || out->dim_np == NULL || out->ready == NULL)
        return IO_NULL_ARGUMENT;
    int32_t nu, ny, np;
    if (sizes(in->variant, &nu, &ny, &np) != 0) return IO_UNKNOWN_VARIANT;

    if (*in->init_flag == 1) {
        *out->dim_nu = nu;
        *out->dim_ny = ny;
        *out->dim_np = np;
        for (int i = 0; i < ny; i++) {
            out->y[i] = 0.0;
            for (int j = 0; j < nu; j++) out->gain[i * nu + j] = (i + 1) + 0.1 * j;
        }
        for (int i = 0; i < np; i++) out->p[i] = 0.5 * i;
        if (out->label != NULL)
            copy_text(out->label, 32, (in->title != NULL && in->title[0] != '\0') ? in->title
                                                                                   : "io");
        if (out->names != NULL) copy_text(out->names, 24, "k;u;y;p");
        *in->counter = 0;
        *out->ready = IO_TRUE_BYTE;
        return IO_OK;
    }
    if (*in->init_flag != 0) return IO_BAD_FLAG;
    if (*out->dim_nu != nu || *out->dim_ny != ny || *out->dim_np != np) return IO_NOT_INITIALISED;
    for (int i = 0; i < ny; i++) {
        double acc = 0.0;
        for (int j = 0; j < nu; j++) acc += out->gain[i * nu + j] * in->u[j];
        out->y[i] = *in->k * acc;
    }
    *in->counter += 1;
    *out->ready = IO_TRUE_BYTE;
    return IO_OK;
}

read_run.py:

"""Read one recorded run with the Python standard library alone.

Usage: python3 read_run.py <run-dir>
"""

import csv
import gzip
import json
import sys
from pathlib import Path


def rows(run: Path):
    """Yield every row of the run as a dict, parts in order, header lines skipped."""
    for part in sorted(run.glob("part-*.csv.gz")):
        with gzip.open(part, "rt", newline="") as f:
            try:
                yield from csv.DictReader(f)
            except EOFError:
                pass  # the open part of a running engine ends at its last flush


def main() -> None:
    run = Path(sys.argv[1])
    meta = json.loads((run / "run.json").read_text())
    every = meta["every"]
    n, gaps, last, prev = 0, 0, None, None
    for row in rows(run):
        cycle = int(row["cycle"])
        if prev is not None and cycle != prev + every:
            gaps += 1  # rows lost between prev and cycle
        prev, last, n = cycle, row, n + 1
    print(
        f"{meta['project']}/{meta['recorder']}: {n} rows, tick {meta['tick_s']} s, "
        f"every {every}, {gaps} gaps"
    )
    print("last row:", last)
    with open(run / "events.csv", newline="") as f:
        for e in csv.DictReader(f):
            print(f"event {e['cycle']} {e['kind']} {e['source']}: {e['detail']}")


if __name__ == "__main__":
    main()

/* A tank drained through an outlet and filled by a pump and a disturbance inflow, with a noisy
 * level sensor; in the recommended shape. */
#ifndef TANK_H
#define TANK_H

#include <stdint.h>

typedef struct {
    double area;     /* m2: cross-section */
    double k_out;    /* m2.5/s: outlet coefficient, outflow = k_out sqrt(level) */
    double q_max;    /* m3/s: pump capacity */
    double level0;   /* m: level at the start */
    double noise_sd; /* m: standard deviation of the sensor noise */
    uint64_t seed;   /* seed of the sensor noise */
} tank_params;

typedef struct {
    double q_set;  /* m3/s: pump flow asked for */
    double q_dist; /* m3/s: disturbance inflow */
} tank_inputs;

typedef struct {
    double level;   /* m: true level */
    double level_m; /* m: measured level */
    double q_pump;  /* m3/s: pump flow, q_set clamped to 0..q_max */
} tank_outputs;

int tank_init(void **h, const tank_params *params, double step_size);
int tank_step(void *h, double time, const tank_inputs *in, tank_outputs *out);
void tank_terminate(void *h);

#endif

analyze.py:

"""Rank the gains of the level-loop campaign by how well each run rejects the disturbance.

    uv run --with taktwerk python examples/campaign/analyze.py examples/campaign/out/gains

For every run: the integral of the absolute level error after the disturbance step (300 s to
700 s, before the sensor freezes), the largest error in that window, and the largest error while
the sensor is frozen (700 s to 760 s) and the minute after. Per case, the runs best first.
"""

from __future__ import annotations

import sys

import taktwerk


def window(run: taktwerk.Run, start: float, end: float) -> list[int]:
    """Row indices with start <= t_engine < end."""
    return [i for i, t in enumerate(run.t_engine) if start <= t < end]


def metrics(run: taktwerk.Run) -> dict[str, float]:
    level, sp = run["plant.level"], run["ctrl.sp[0]"]
    dt = run.manifest["tick_s"]
    after = window(run, 300.0, 700.0)
    frozen = window(run, 700.0, 820.0)
    return {
        "iae": sum(abs(level[i] - sp[i]) for i in after) * dt,
        "max_err": max(abs(level[i] - sp[i]) for i in after),
        "max_err_frozen": max(abs(level[i] - sp[i]) for i in frozen),
    }


def main(out: str) -> None:
    campaign = taktwerk.read_campaign(out)
    rows = []
    for r in campaign:
        if not r.ok:
            print(f"{r.name}: {r.status}, skipped", file=sys.stderr)
            continue
        rows.append((r, metrics(r.read())))
    for case in dict.fromkeys(r.case for r, _ in rows):
        print(f"\n{case}")
        print(f"  {'run':<18} {'kp':>5} {'ki':>6} {'IAE [m s]':>10} {'max err [m]':>12} {'frozen [m]':>11}")
        for r, m in sorted((x for x in rows if x[0].case == case), key=lambda x: x[1]["iae"]):
            print(
                f"  {r.name:<18} {r.values['ctrl.kp']:>5} {r.values['ctrl.ki']:>6} "
                f"{m['iae']:>10.3f} {m['max_err']:>12.4f} {m['max_err_frozen']:>11.4f}"
            )


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(f"usage: {sys.argv[0]} <campaign output directory>")
    main(sys.argv[1])
